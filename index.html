        // Simplistic standard mathematical algorithm for solar prayer times (USNO approx)
        function calcSolarPrayerTimes(lat, lng, tz, date) {
            const rad = Math.PI/180, deg = 180/Math.PI;
            // Day of year
            const start = new Date(date.getFullYear(), 0, 0);
            const diff = date - start;
            const day = Math.floor(diff / 1000 / 60 / 60 / 24);
            
            // Solar Declination & Eq of Time approximation
            const b = (2 * Math.PI * (day - 81)) / 365;
            const dec = 23.45 * Math.sin(b);
            const eqT = 9.87 * Math.sin(2 * b) - 7.53 * Math.cos(b) - 1.5 * Math.sin(b);
            
            // Zuhr transit
            const zuhrDec = 12 + tz - (lng / 15) - (eqT / 60);
            
            // Hour angle calculation func
            const calcHA = (angleDeg) => {
                const cosHA = (Math.sin(-angleDeg*rad) - Math.sin(lat*rad)*Math.sin(dec*rad)) / (Math.cos(lat*rad)*Math.cos(dec*rad));
                if(cosHA > 1 || cosHA < -1) return 0; // extreme lats
                return Math.acos(cosHA) * deg / 15;
            };

            // MWL: Fajr 18, Isha 17. Sunrise/Sunset 0.833
            const fajrHA = calcHA(18);
            const ishaHA = calcHA(17);
            const sunsetHA = calcHA(0.833);
            
            // Asr Shafi (shadow length = 1 + shadow at noon)
            const altNoon = Math.abs(lat - dec);
            const asrAlt = Math.atan(1 / (1 + Math.tan(altNoon*rad))) * deg;
            const asrHA = calcHA(asrAlt);

            const format = (decHours) => {
                let h = Math.floor(decHours);
                let m = Math.floor((decHours - h) * 60);
                let dateObj = new Date(date);
                dateObj.setHours(h, m, 0, 0);
                
                let ampm = h >= 12 ? 'PM' : 'AM';
                let h12 = h % 12; if(h12 === 0) h12 = 12;
                let mStr = m < 10 ? '0'+m : m;
                return { str: `${h12}:${mStr} ${ampm}`, dateObj: dateObj, status: 'Calculated' };
            };

            return {
                fajr: format(zuhrDec - fajrHA),
                dhuhr: format(zuhrDec),
                asr: format(zuhrDec + asrHA),
                maghrib: format(zuhrDec + sunsetHA),
                isha: format(zuhrDec + ishaHA)
            };
        }
